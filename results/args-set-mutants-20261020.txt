def real_not_now():
    """H.not_now(), but a driver inside another harness's sandbox (under the temp dir's stubs.* directories, the copies
    the parallel mutants run) does not count: it starts the sandbox's stub service, not strata-q3."""
    import tempfile
    sandbox = os.path.join(tempfile.gettempdir(), "stubs.")
    drivers = (b"bench/ab_engine.sh", b"bench/ab_model_run.sh", b"bench/switch_model.sh", b"ab-ops/switch_model.sh")
    real = [pid for pid, cmd in H._cmdlines()
            if any(part.endswith(d) and not part.startswith(sandbox.encode()) for part in cmd[:3] for d in drivers)]
    if real:
        return f"an experiment driver runs (pids {real})"
    if H._active("strata-ab-rescue.timer"):
        return "strata-ab-rescue.timer is armed"
    r = subprocess.run(["systemctl", "is-active", "strata-q3.service"], capture_output=True, text=True, timeout=10)
    if r.stdout.strip() in ("activating", "reloading"):
        return "strata-q3 is starting"
    return None
