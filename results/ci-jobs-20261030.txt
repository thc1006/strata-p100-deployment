"""Compare the outcome rows (BEFORE-ROW/AFTER-ROW lines of a CI log): argv log [before-prefix after-prefix]."""
import sys
log = open(sys.argv[1], errors="replace").read().splitlines()
pa, pb = (sys.argv[2], sys.argv[3]) if len(sys.argv) > 3 else ("BEFORE-ROW ", "AFTER-ROW ")
def rows(prefix):
    out = []
    for l in log:
        i = l.find(prefix)
        if i >= 0 and l[i - 1] == " " and not l[i:].startswith(prefix + "/"):
            out.append(l[i + len(prefix):])
    return out
b, a = rows(pa), rows(pb)
ids = lambda rs: {r.split("\t")[1] for r in rs if r.startswith("started\t")}
bi, ai = ids(b), ids(a)
print("before: rows", len(b), "started ids", len(bi), "| after: rows", len(a), "started ids", len(ai))
print("ids only before:", sorted(bi - ai))
print("ids only after:", len(ai - bi)); [print("  ", x) for x in sorted(ai - bi)]
def key(r): return r.split("\t")[1].split(" ")[0]
common = lambda rs: sorted(r for r in rs if key(r) in bi and not r.startswith("testsRun"))
cb, ca = common(b), common(a)
print("rows of the common tests equal:", cb == ca)
for r in sorted(set(cb) ^ set(ca)): print("  DIFF", "before" if r in cb else "after", r)
print("testsRun", [r for r in b if r.startswith("testsRun")], [r for r in a if r.startswith("testsRun")])
import collections
print("before kinds", collections.Counter(r.split("\t")[0] for r in b))
print("after kinds", collections.Counter(r.split("\t")[0] for r in a))

import re, collections, sys
rows = []
for l in open(sys.argv[1], errors="replace"):
    m = re.search(r" TIME (w\d+) +([\d.]+) +([\d.]+) (\S+)", l)
    if m: rows.append((m[1], float(m[2]), float(m[3]), m[4]))
    m = re.search(r"Ran (\d+) tests in ([\d.]+)s", l)
    if m: print(l[l.index("Ran "):].strip())
end = collections.defaultdict(float); busy = collections.defaultdict(float)
for w, a, d, i in rows:
    end[w] = max(end[w], a + d); busy[w] += d
print(len(rows), "tests;", " ".join(f"{w}:end {end[w]:.0f}" for w in sorted(end)))
for w, a, d, i in sorted(rows, key=lambda r: -r[2])[:6]: print(f"  {w} start {a:6.1f} dur {d:6.1f} {i}")
mods = collections.defaultdict(float)
for w, a, d, i in rows: mods[i.split(".")[0]] += d
print({k: round(v) for k, v in sorted(mods.items(), key=lambda x: -x[1])[:6]}, "sum", round(sum(r[2] for r in rows)))
