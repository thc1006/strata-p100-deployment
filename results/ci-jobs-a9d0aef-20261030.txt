"""Compare the outcome rows (BEFORE-ROW/AFTER-ROW lines of a CI log): argv log [before-prefix after-prefix]."""
import sys
log = open(sys.argv[1], errors="replace").read().splitlines()
pa, pb = (sys.argv[2], sys.argv[3]) if len(sys.argv) > 3 else ("BEFORE-ROW ", "AFTER-ROW ")
def rows(prefix):
    out = []
    for l in log:
        i = l.find(prefix)
        if i >= 0 and l[i - 1] == " " and not l[i:].startswith(prefix + "/"):
            out.append(l[i + len(prefix):])
    return out
b, a = rows(pa), rows(pb)
ids = lambda rs: {r.split("\t")[1] for r in rs if r.startswith("started\t")}
bi, ai = ids(b), ids(a)

print("ids only before:", sorted(bi - ai))
print("ids only after:", len(ai - bi)); [print("  ", x) for x in sorted(ai - bi)]
def key(r): return r.split("\t")[1].split(" ")[0]
common = lambda rs: sorted(r for r in rs if key(r) in bi and not r.startswith("testsRun"))

import datetime, json, re, subprocess, sys
r1_jobs, r1_log, r2_jobs, r2_log, compare = sys.argv[1:6]


def step(jobs_file):
    job = json.load(open(jobs_file))["jobs"][0]
    s = next(s for s in job["steps"] if s["name"].startswith("scripts/test.sh"))
    a, b = (datetime.datetime.strptime(s[k], "%Y-%m-%dT%H:%M:%SZ") for k in ("started_at", "completed_at"))
    d = int((b - a).total_seconds())
    return (job["run_id"], job["head_sha"][:7], f"step '{s['name']}' ({s['conclusion']}) from {a:%H:%M:%S} to "
            f"{b:%H:%M:%S} ({d // 60} min {d % 60} s)")


def lines(log, pattern):
    return [l.rstrip("\n")[:200] for l in open(log, errors="replace")
            if re.match(r"\d{4}-\d\d-\d\dT", l) and re.search(pattern, l)]
