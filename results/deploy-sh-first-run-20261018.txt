#!/usr/bin/env bash
# End-to-end run of scripts/deploy.sh against the real GitHub repository, in a scratch clone (never the live checkout).
# The clone starts at OLD, which has no deploy.sh; the deploy.sh of origin/main runs from an ignored file next to where
# the tracked one will be, and deploys origin/main into the clone.  Usage: e2e.sh <scratch dir> <OLD commit>
set -u -o pipefail
E=$1; OLD=$2
rm -rf "$E" && mkdir -p "$E" || exit 1
git clone -q https://github.com/thc1006/strata-p100-deployment.git "$E/live" || exit 1
cd "$E/live" || exit 1
NEW=$(git rev-parse HEAD)
git show "$NEW:scripts/deploy.sh" > "$E/deploy.sh" || exit 1
git reset -q --hard "$OLD" || exit 1
cp "$E/deploy.sh" scripts/deploy-e2e.sh && chmod 755 scripts/deploy-e2e.sh && echo scripts/deploy-e2e.sh >> .git/info/exclude
echo "clone at $(git log -1 --format='%h %s'); origin/main is $(git log -1 --format='%h %s' "$NEW")"
echo "status: [$(git status --porcelain --untracked-files=all)]"
echo "== scripts/deploy-e2e.sh"; time scripts/deploy-e2e.sh; echo "exit $?"
echo "== scripts/deploy-e2e.sh --apply"; time scripts/deploy-e2e.sh --apply; echo "exit $?"
echo "== after"; echo "HEAD $(git log -1 --format='%h %s'), status: [$(git status --porcelain --untracked-files=all)]"
cmp -s scripts/deploy.sh "$E/deploy.sh" && echo "scripts/deploy.sh is the tracked file of $(git rev-parse --short "$NEW")"
echo "== scripts/deploy.sh --apply (the tracked one)"; scripts/deploy.sh --apply; echo "exit $?"
