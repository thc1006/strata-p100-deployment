def build(mid, reps):
    text = SRC
    for old, new in reps:
        if text.count(old) != 1:
            return None, f"pattern matches {text.count(old)} times: {old[:60]!r}"
        text = text.replace(old, new)
    d = HERE / "m" / mid
    shutil.rmtree(d, ignore_errors=True); (d / "scripts").mkdir(parents=True); (d / "tests").mkdir()
    (d / "scripts" / "deploy.sh").write_text(text); (d / "scripts" / "deploy.sh").chmod(0o755)
    (d / "tests" / "test_deploy.py").write_text(TEST)
    return d, None

def run(mid, desc, reps):
    d, err = build(mid, reps)
    if err:
        return mid, desc, "INVALID", err, 0.0
    s = time.monotonic()
    try:
        r = subprocess.run(["nice", "-n", "19", "python3", "-B", "-m", "unittest", "discover", "-f", "-s", "tests", "-p", "test_deploy.py"],
                           cwd=d, capture_output=True, text=True, timeout=600)
        verdict = "survived" if r.returncode == 0 else "caught"
        why = "" if r.returncode == 0 else next((l for l in r.stderr.splitlines() if l.startswith(("FAIL:", "ERROR:"))), "")
    except subprocess.TimeoutExpired:
        verdict, why = "caught", "timeout"
    return mid, desc, verdict, why, time.monotonic() - s

base = run("M00", "no change", [])
print(f"{base[0]} {base[1]}: {base[2]} {base[3]} ({base[4]:.0f} s)", flush=True)
if base[2] != "survived": sys.exit("the unchanged script fails its tests")
with cf.ThreadPoolExecutor(WORKERS) as ex:
    results = list(ex.map(lambda m: run(*m), MUTANTS))
for mid, desc, verdict, why, secs in results:
    print(f"{mid} {verdict:8} {desc}{'  -- ' + why if why else ''}")
c = sum(r[2] == "caught" for r in results); s = sum(r[2] == "survived" for r in results); i = sum(r[2] == "INVALID" for r in results)
print(f"{len(results)} mutants: {c} caught, {s} survived, {i} invalid")
