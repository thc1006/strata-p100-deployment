"""The first metadata keys of a GGUF file, up to and including the architecture's context_length (standard library)."""
import struct, sys
f = open(sys.argv[1], "rb")
def rd(fmt):
    return struct.unpack("<" + fmt, f.read(struct.calcsize(fmt)))
assert f.read(4) == b"GGUF"
ver, = rd("I")
nt, nkv = rd("QQ")
SZ = {0: "B", 1: "b", 2: "H", 3: "h", 4: "I", 5: "i", 6: "f", 7: "?", 10: "Q", 11: "q", 12: "d"}
print(f"GGUF version {ver}, {nt} tensors, {nkv} metadata keys")
for _ in range(nkv):
    n, = rd("Q")
    key = f.read(n).decode()
    t, = rd("I")
    if t in SZ:
        v = rd(SZ[t])[0]
    elif t == 8:
        n, = rd("Q")
        v = f.read(n).decode("utf-8", "replace")
    else:
        print(f"{key}: type {t}, not read; stopping")
        break
    print(f"{key} = {v}")
    if key.endswith(".context_length"):
        break
