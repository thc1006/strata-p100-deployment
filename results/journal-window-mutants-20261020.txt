== mutate_journal.py
"""Broken copies of the journal windows of bench/ab_model_run.sh and bench/ab_engine.sh, each in a copy of the worktree,
against their harnesses (tests/ab_model_run_stubs.sh, tests/ab_engine_stubs.sh, run by tests/harness_run.py, three
copies at a time).  Usage: python3 -B mutate_journal.py <worktree>"""
import concurrent.futures, difflib, hashlib, shutil, subprocess, sys, tempfile
from pathlib import Path

W = Path(sys.argv[1])
sys.path.insert(0, str(W / "tests"))
import harness_run as H  # noqa: E402

head = lambda f: subprocess.run(["git", "-C", str(W), "show", f"HEAD:{f}"], capture_output=True, text=True, check=True).stdout
R = "bench/ab_model_run.sh"
E = "bench/ab_engine.sh"
M = [
    (R, "the script before the fix (HEAD)", None, None),
    (R, "--after-cursor for the service's window", 'journalctl -u $UNIT --cursor "$cur"', 'journalctl -u $UNIT --after-cursor "$cur"'),
    (R, "--after-cursor for the kernel's window", 'journalctl -k --cursor "$cur"', 'journalctl -k --after-cursor "$cur"'),
    (R, "the journal's last entry instead of the mark", 'c=$(sudo -n journalctl -t "$id" -n 1 --show-cursor', 'c=$(sudo -n journalctl -n 1 --show-cursor'),
    (R, "no refusal without a mark", '[ -n "$cur" ] || refuse "cannot mark the journal', ': || refuse "cannot mark the journal'),
    (R, "one look for the mark, no wait", "  for i in $(seq 100); do\n    c=$(sudo -n journalctl -t", "  for i in 1; do\n    c=$(sudo -n journalctl -t"),
    (E, "the script before the fix (HEAD)", None, None),
    (E, "--after-cursor for the test unit's window", 'journalctl -u "$UNIT" --cursor "$cur"', 'journalctl -u "$UNIT" --after-cursor "$cur"'),
    (E, "--after-cursor for the kernel's window", 'journalctl -k --cursor "$cur"', 'journalctl -k --after-cursor "$cur"'),
    (E, "no refusal without a mark", '[ -n "$cur" ] || { say "cannot mark the journal', ': || { say "cannot mark the journal'),
]
HARNESS = {R: "tests/ab_model_run_stubs.sh", E: "tests/ab_engine_stubs.sh"}


def one(m):
    f, name, old, new = m
    src = (W / f).read_text()
    text = head(f) if old is None else src.replace(old, new)
    if old is not None and src.count(old) != 1:
        return m, None, [f"PATTERN found {src.count(old)} times"], ""
    d = Path(tempfile.mkdtemp(prefix="mutj."))
    try:
        for sub in ("bench", "scripts", "tools", "systemd", "config", "tests"):
            shutil.copytree(W / sub, d / sub, ignore=shutil.ignore_patterns("__pycache__"))
        (d / f).write_text(text)
        r = H.run(d / HARNESS[f], d, 1500)
        fails = [l[5:] for l in r["out"].splitlines() if l.startswith("FAIL ")]
        killed = r["status"] != 0 or bool(fails)
        diff = "".join(difflib.unified_diff(src.splitlines(True), text.splitlines(True), f, f"copy: {name}")).rstrip("\n")
        return m, killed, [(x.splitlines() or [""])[0].replace(str(d), "<tmp>")[:150] for x in fails], diff
    finally:
        shutil.rmtree(d, ignore_errors=True)


sha = lambda p: hashlib.sha256((W / p).read_bytes()).hexdigest()
ver = lambda *c: subprocess.run(c, capture_output=True, text=True).stdout.split("\n")[0]
print("# The journal windows of bench/ab_model_run.sh and bench/ab_engine.sh against broken copies of them, 2026-10-20,")
print("# after the fix that reads each run's journal and kernel lines from an entry the script writes (logger) with")
print("# --cursor, instead of --after-cursor from the journal's last entry.")
print("#")
print("# Each copy has one deliberate change, made by the text replacements in the runner at the end of this file; its")
print("# diff against the script is below.  The script's harness runs on a copy of the worktree that holds it, through")
print("# tests/harness_run.py; \"killed\" means that the harness failed, and the checks that failed are listed.  The stub")
print("# journalctl answers --after-cursor as systemd 249 does after a cursor whose own entry does not match the filter")
print("# (results/journalctl-after-cursor-20261020.txt): without the first matching line.")
print(f"# Machine: <host>, {ver('lsb_release', '-ds')}, {ver('python3', '-V')}, bash {ver('bash', '-c', 'echo $BASH_VERSION')}.  "
      "Temporary directories are written <tmp>.")
print("# Inputs (sha256): " + "\n#                  ".join(f"{p} {sha(p)}" for p in (R, E, "tests/ab_model_run_stubs.sh", "tests/ab_engine_stubs.sh", "tests/harness_run.py")))
with concurrent.futures.ThreadPoolExecutor(3) as ex:
    results = list(ex.map(one, M))
for f in (R, E):
    rs = [x for x in results if x[0][0] == f]
    print(f"\n== {f}: {sum(1 for x in rs if x[1])} of {len(rs)} killed   (python3 -B mutate_journal.py <worktree>)")
    for (_, name, _, _), killed, fails, diff in rs:
        print(f"\n-- {name}: {'killed' if killed else 'SURVIVED'}, {len(fails)} failed check{'' if len(fails) == 1 else 's'}")
        for x in fails[:6]:
            print(f"FAIL {x}")
        if len(fails) > 6:
            print(f"... and {len(fails) - 6} more")
        if diff:
            print(diff)
print("\n== mutate_journal.py")
