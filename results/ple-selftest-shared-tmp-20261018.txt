## race.sh
#!/bin/bash
# Two --selftest processes of the 0.1.45 ple_reader_test at once, as two ctest runs would start them.
# shared: both get the same temp directory (TMPDIR, which std::filesystem::temp_directory_path() returns).
# control: each gets its own directory.
set -u
cd "$(dirname "$0")"
bin=./ple_reader_test_477
for mode in shared control; do
  for off in 0 0.3 1 2 4 6; do
    mkdir -p "r/$mode-$off/a" "r/$mode-$off/b"
    da=r/$mode-$off/a; db=r/$mode-$off/b; [ $mode = shared ] && db=$da
    ( TMPDIR=$PWD/$da nice -n 19 $bin --selftest > r/$mode-$off/out_a.txt 2>&1; echo $? > r/$mode-$off/rc_a ) &
    sleep $off
    ( TMPDIR=$PWD/$db nice -n 19 $bin --selftest > r/$mode-$off/out_b.txt 2>&1; echo $? > r/$mode-$off/rc_b ) &
    wait
    for p in a b; do
      printf '%-8s offset %-4s %s rc=%-4s %s\n' $mode $off $p "$(cat r/$mode-$off/rc_$p)" "$(grep -m1 -E 'FAIL|cannot|error|differ|Bus|Segm' r/$mode-$off/out_$p.txt | cut -c1-150)"
    done
  done
done

## reaper.sh (started while race.sh ran the offset 2 pair)
#!/bin/bash
# Kill a ple_reader_test_477 run that has been alive for more than 180 s (a normal run takes 15-30 s here),
# after recording its elapsed and CPU time, its threads' wait channels and its open table file.
set -u
log="$1"
while pgrep -f '[r]ace.sh' >/dev/null; do
  for p in $(pgrep -f '[p]le_reader_test_477 --selftest'); do
    et=$(ps -o etimes= -p $p 2>/dev/null | tr -d ' '); [ -n "$et" ] || continue
    if [ "$et" -gt 180 ]; then
      { echo "$(date +%T) pid $p: $(ps -o etimes=,time=,stat= -p $p) cwd-out=$(readlink /proc/$p/fd/1)";
        echo "  threads: $(ls /proc/$p/task | wc -l), wchan: $(cat /proc/$p/task/*/wchan 2>/dev/null | sort | uniq -c | tr '\n' ' ')";
        echo "  fd: $(ls -l /proc/$p/fd 2>/dev/null | grep -o '/[^ ]*gguf.*\|/[^ ]*\.bin.*' | tr '\n' ' ')"; } >> "$log"
      kill -TERM $p
    fi
  done
  sleep 10
done
echo "$(date +%T) race.sh finished" >> "$log"
