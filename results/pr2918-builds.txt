#!/bin/bash
# The HIP engine and ple_reader_test, configured the way Strata's setup.py does it (build_engine_hip), plus the test.
# --gcc-install-dir: this machine has GCC 12's runtime but only GCC 11's C++ headers, and clang would pick 12.
set -u
S=<work>
W=/home/thc1006/dev/strata-0.1.45-locked
CB=/home/thc1006/dev/strata/.venv/lib/python3.10/site-packages/cmake/data/bin
NINJA=/home/thc1006/dev/strata/.venv/bin/ninja
ROOT=$($S/tc/rocm-venv/bin/rocm-sdk path --root)
LIBS=$S/tc/rocm-venv/lib/python3.10/site-packages/_rocm_sdk_libraries_gfx110X_dgpu
BITCODE=$ROOT/lib/llvm/amdgcn/bitcode
export HIP_PLATFORM=amd HIP_COMPILER=clang HIP_RUNTIME=rocclr ROCM_PATH=$ROOT HIP_PATH=$ROOT
export LD_LIBRARY_PATH=$ROOT/lib:$LIBS/lib${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}
export PATH=$ROOT/bin:$ROOT/llvm/bin:$PATH
cd $W || exit 1
echo "head: $(git rev-parse HEAD)"
echo "rocm: $($S/tc/rocm-venv/bin/rocm-sdk version 2>&1 | head -1)"
$ROOT/llvm/bin/clang++ --version | head -1
nice -n 19 $CB/cmake -G Ninja -DCMAKE_MAKE_PROGRAM=$NINJA -S $W -B $W/build-hip -DCMAKE_BUILD_TYPE=Release \
  -DSTRATA_ENABLE_HIP=ON -DSTRATA_ENABLE_CUDA=OFF -DSTRATA_BUILD_TESTS=OFF -DSTRATA_PREFILL_MMQ=ON \
  -DSTRATA_Q6K_EXPERTS=ON -DSTRATA_MMQ_KQUANTS=ON -DCMAKE_HIP_ARCHITECTURES=gfx1100 \
  -DCMAKE_HIP_COMPILER=$ROOT/llvm/bin/clang++ -DCMAKE_HIP_COMPILER_ROCM_ROOT=$ROOT \
  "-DCMAKE_PREFIX_PATH=$ROOT;$LIBS" "-DCMAKE_HIP_FLAGS=--rocm-path=$ROOT --rocm-device-lib-path=$BITCODE --gcc-install-dir=/usr/lib/gcc/x86_64-linux-gnu/11" \
  -DSTRATA_GGML_DIR=$W/third_party/llama.cpp || { echo "CONFIGURE FAILED"; exit 1; }
time nice -n 19 $CB/cmake --build $W/build-hip --target strata ple_reader_test -j 2
rc=$?
echo "BUILD rc=$rc"
exit $rc
