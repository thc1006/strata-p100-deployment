== gguf_head.py
#!/usr/bin/env python3
"""Read a GGUF file's header (local path or https URL, by HTTP range requests) and print its tensors as JSON lines:
name, type, dims, offset, bytes (from the offsets; the last tensor's from the file size).  Standard library only."""
import json, re, struct, sys, urllib.request

TYPES = {0: "F32", 1: "F16", 2: "Q4_0", 3: "Q4_1", 6: "Q5_0", 7: "Q5_1", 8: "Q8_0", 9: "Q8_1", 10: "Q2_K", 11: "Q3_K",
         12: "Q4_K", 13: "Q5_K", 14: "Q6_K", 15: "Q8_K", 16: "IQ2_XXS", 17: "IQ2_XS", 18: "IQ3_XXS", 19: "IQ1_S",
         20: "IQ4_NL", 21: "IQ3_S", 22: "IQ2_S", 23: "IQ4_XS", 24: "I8", 25: "I16", 26: "I32", 27: "I64", 28: "F64",
         29: "IQ1_M", 30: "BF16", 34: "TQ1_0", 35: "TQ2_0", 39: "MXFP4"}


class Src:
    def __init__(self, where):
        self.where, self.buf, self.pos, self.size = where, b"", 0, None
        if not where.startswith("http"):
            self.f = open(where, "rb")
            self.f.seek(0, 2); self.size = self.f.tell()

    def need(self, n):
        while self.pos + n > len(self.buf):
            want = max(len(self.buf) * 2, 1 << 22, self.pos + n)
            if self.where.startswith("http"):
                req = urllib.request.Request(self.where, headers={"Range": f"bytes={len(self.buf)}-{want - 1}",
                                                                  "User-Agent": "gguf-head/1"})
                with urllib.request.urlopen(req, timeout=120) as r:
                    cr = r.headers.get("Content-Range") or ""
                    m = re.search(r"/(\d+)$", cr)
                    if m:
                        self.size = int(m.group(1))
                    chunk = r.read()
                    if r.status == 200:            # no range support: the whole body came back
                        self.buf = chunk; self.size = len(chunk); continue
            else:
                self.f.seek(len(self.buf)); chunk = self.f.read(want - len(self.buf))
            if not chunk:
                raise EOFError(self.where)
            self.buf += chunk

    def read(self, n):
        self.need(n); b = self.buf[self.pos:self.pos + n]; self.pos += n; return b

    def u32(self): return struct.unpack("<I", self.read(4))[0]
    def u64(self): return struct.unpack("<Q", self.read(8))[0]
    def s(self): return self.read(self.u64()).decode("utf-8", "replace")


SCALAR = {0: "<B", 1: "<b", 2: "<H", 3: "<h", 4: "<I", 5: "<i", 6: "<f", 7: "<?", 10: "<Q", 11: "<q", 12: "<d"}


def value(src, t):
    if t in SCALAR:
        fmt = SCALAR[t]; return struct.unpack(fmt, src.read(struct.calcsize(fmt)))[0]
    if t == 8:
        return src.s()
    if t == 9:
        et, n = src.u32(), src.u64()
        if et in SCALAR:                                   # skip numeric arrays quickly
            src.read(struct.calcsize(SCALAR[et]) * n); return f"<array {n}>"
        return [value(src, et) for _ in range(n)] if n < 64 else ([value(src, et) for _ in range(n)] and f"<array {n}>")
    raise ValueError(f"kv type {t}")


def main(where):
    src = Src(where)
    assert src.read(4) == b"GGUF", "not GGUF"
    ver, nt, nkv = src.u32(), src.u64(), src.u64()
    kv = {}
    for _ in range(nkv):
        k = src.s(); t = src.u32(); kv[k] = value(src, t)
    align = int(kv.get("general.alignment", 32))
    tensors = []
    for _ in range(nt):
        name = src.s(); nd = src.u32(); dims = [src.u64() for _ in range(nd)]; t = src.u32(); off = src.u64()
        tensors.append({"name": name, "type": TYPES.get(t, f"type{t}"), "dims": dims, "offset": off})
    data_start = (src.pos + align - 1) // align * align
    if src.size is None and where.startswith("http"):
        src.need(1)
    order = sorted(range(len(tensors)), key=lambda i: tensors[i]["offset"])
    for a, b in zip(order, order[1:] + [None]):
        end = tensors[b]["offset"] if b is not None else src.size - data_start
        tensors[a]["bytes"] = end - tensors[a]["offset"]
    meta = {k: v for k, v in kv.items() if not isinstance(v, str) or len(v) < 200}
    meta = {k: v for k, v in meta.items() if not k.startswith("tokenizer.ggml.")}
    print(json.dumps({"file": where.rsplit("/", 1)[-1], "version": ver, "tensors": nt, "size": src.size,
                      "data_start": data_start, "meta": meta}, default=str))
    for t in tensors:
        print(json.dumps(t))


if __name__ == "__main__":
    main(sys.argv[1])

"""Where each candidate model would live on this machine: VRAM per GPU (best contiguous layer split), the experts that
spill to pinned RAM, and the RAM left above the 8 GiB floor with and without longer context (KV streaming).
Capacities are measured on the running engine (log of the 2026-10-18 02:26 start, /metrics at 21:4x)."""
import itertools, json, re, sys

GI = 2 ** 30
L = 48
# experts each GPU holds now (exact, from the GGUF headers = the engine log) + its VRAM free beyond the 1536 MiB
# reserve (/metrics mem_used against the engine's 15.89 GiB per card)
NOW_EXPERTS = [9.41, 9.88, 11.43, 9.25]
SPARE = [1111 / 1024, 797 / 1024, 420 / 1024, 228 / 1024]
TWEAKS = {"as deployed": [0, 0, 0, 0],
          "+kv-resident 20480, reserve 1024": [0.145 + 0.5] * 4,
          "+ also prefill 1024 (est.)": [0.145 + 0.5 + 0.55] * 4}
HEADROOM = 18.6 - 8.0          # MemAvailable now minus the 8 GiB floor
ARC_CUT = 7.47 - 2.0           # if the ARC limit went from 8 to 2 GiB
KV = {"32K (now)": 0.0, "128K x4 int8": 4 * 131072 * 13.7e3 / GI, "128K x4 k8v4": 4 * 131072 * 10.8e3 / GI,
      "256K x4 k8v4": 4 * 262144 * 10.8e3 / GI}


def load(prefix, files):
    exp, dense, ple = [0] * L, [0] * L, 0
    head = 0
    for fn in files:
        for i, line in enumerate(open(fn)):
            if i == 0:
                continue
            t = json.loads(line)
            m = re.match(r"blk\.(\d+)\.", t["name"])
            if "_exps" in t["name"]:
                exp[int(m.group(1))] += t["bytes"]
            elif t["name"].startswith("per_layer_token_embd"):
                ple += t["bytes"]
            elif m:
                dense[int(m.group(1))] += t["bytes"]
            elif t["name"].startswith("output"):
                head += t["bytes"]
    return [e / GI for e in exp], [d / GI for d in dense], head / GI, ple / GI


def best_split(exp, dense, head, cap):
    best = None
    for a, b, c in itertools.combinations(range(1, L), 3):
        spans = [(0, a), (a, b), (b, c), (c, L)]
        spill = 0
        for g, (x, y) in enumerate(spans):
            need = sum(exp[x:y]) + sum(dense[x:y]) - BASE_DENSE[g] + (head - BASE_HEAD if g == 3 else 0)
            spill += max(0.0, need - cap[g])
        if best is None or spill < best[0]:
            best = (spill, (a, b, c))
    return best


base = load("iq3_xxs", ["iq3_xxs-1.jsonl", "iq3_xxs-2.jsonl"])
SPL = [(0, 13), (13, 26), (26, 38), (38, 48)]
BASE_DENSE = [sum(base[1][x:y]) for x, y in SPL]
BASE_HEAD = base[2]
MODELS = {"IQ3_XXS (now)": "iq3_xxs", "GSQ-RCO IQ3_S": "IQ3_S", "Unsloth UD-IQ4_XS": "UD-IQ4_XS",

          "Unsloth UD-Q4_K_XL": "UD-Q4_K_XL", "bartowski Q4_K_M": "bartowski-Q4_K_M"}
import glob
rows = []
for name, pre in MODELS.items():
    files = sorted(glob.glob(f"{pre}-*.jsonl"))
    exp, dense, head, ple = load(pre, files)
    out = {"model": name, "experts": sum(exp), "dense_delta": sum(dense) + head - sum(BASE_DENSE) - BASE_HEAD,
           "ple": ple}
    for tname, tw in TWEAKS.items():
        cap = [NOW_EXPERTS[g] + SPARE[g] + tw[g] for g in range(4)]
        spill, split = best_split(exp, dense, head, cap)
        out[tname] = (spill, split)
    rows.append(out)
print(f"VRAM for experts now: {sum(NOW_EXPERTS) + sum(SPARE):.2f} GiB; RAM headroom above the 8 GiB floor: "
      f"{HEADROOM:.1f} GiB (+{ARC_CUT:.1f} if ARC 8->2 GiB)")
print("KV streaming pinned RAM, 4 slots: " + ", ".join(f"{k} {v:.1f} GiB" for k, v in KV.items()))
for r in rows:
    print(f"\n{r['model']}: experts {r['experts']:.2f} GiB, dense+head vs now {r['dense_delta']:+.2f} GiB, PLE {r['ple']:.2f} GiB")
    for tname in TWEAKS:
        spill, split = r[tname]
        res = 100 * (1 - spill / r["experts"])
        stream32 = 4 * 32768 * 13.7e3 / GI if "kv-resident" in tname or "prefill" in tname else 0.0
        extra_ple = r["ple"] - 26.82
        left = {k: HEADROOM - spill - (v if v else stream32) - extra_ple for k, v in KV.items()}
        print(f"  {tname:34} best split {split}: spill {spill:5.2f} GiB to RAM ({res:5.1f}% of expert bytes in VRAM) | "
              "RAM left above floor: " + ", ".join(f"{k} {v:+.1f}/{v + ARC_CUT:+.1f}" for k, v in left.items()))

"""Routing mass the GPUs would hold for each candidate: Strata's profile (data/expert-profile.bin: the ranked pairs

counts = [list(struct.unpack(f"<{ne}i", b[24 + n * 4 + l * ne * 4: 24 + n * 4 + (l + 1) * ne * 4])) for l in range(nl)]
tot = sum(map(sum, counts))
print(f"profile: {nl}x{ne}, {tot:,} routing events; per-layer sums equal: {len({sum(c) for c in counts}) == 1}")
# how skewed: share of routing held by the top x% of each layer's experts
for frac in (0.5, 0.6, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95):
    k = int(ne * frac)
    print(f"  top {frac:.0%} of each layer's experts: {sum(sum(sorted(c, reverse=True)[:k]) for c in counts) / tot:.2%} of routing")

def per_layer(pre):
    exp, dense, head = [0.0] * nl, [0.0] * nl, 0.0
    for fn in sorted(glob.glob(f"{pre}-*.jsonl")):
        for i, line in enumerate(open(fn)):
            if i:
                t = json.loads(line); m = re.match(r"blk\.(\d+)\.", t["name"])
                if "_exps" in t["name"]: exp[int(m.group(1))] += t["bytes"] / GI
                elif m and not t["name"].startswith("per_layer"): dense[int(m.group(1))] += t["bytes"] / GI
                elif t["name"].startswith("output"): head += t["bytes"] / GI
    return exp, dense, head

def held(exp, dense, head, spans, cap):
    hit = 0
    for g, (x, y) in enumerate(spans):
        room = cap[g] - (sum(dense[x:y]) - BD[g]) - ((head - bhead) if g == 3 else 0)
        pairs = sorted(((counts[l][e], l) for l in range(x, y) for e in range(ne)), reverse=True)
        for c, l in pairs:
            sz = exp[l] / ne
            if room < sz:
                continue
            room -= sz; hit += c
    return hit / tot
