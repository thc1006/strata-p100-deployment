#!/bin/bash
# rr [-m MEM] [-c CPUS] [-n NAME] [--sysbus] [--home] -- COMMAND...
# --home: the sandbox's ~/rebuild-rehearsal/root/home/thc1006 is all of /home/thc1006 inside (production's home is not
# visible at all, except the two read-only views and ~/rebuild-rehearsal itself), for installers that need to write
# into /home/thc1006 (the CUDA runfile does).
# One rehearsal job: its own capped user scope (MemoryMax, no swap, oom_score_adj +500, nice 19, SCHED_IDLE, ionice
# idle, CPU affinity), inside a bubblewrap namespace in which
#   - the whole root file system is read-only;
#   - /home/thc1006/dev, /home/thc1006/cuda-12.9, /home/thc1006/downloads and /tmp are directories of the sandbox
#     (so hard-coded production paths resolve into ~/rebuild-rehearsal/root/...);
#   - production's ~/dev and ~/cuda-12.9 are visible only read-only, at ~/rebuild-rehearsal/prod/{dev,cuda-12.9};
#   - /dev is a minimal one without /dev/nvidia* (no GPU can be reached);
#   - its own PID namespace and /proc (no /proc/<pid>/root of a host process), an empty /run/user/1003 (no user bus,
#     so no systemctl --user or systemd-run from inside), and no system bus unless --sysbus is given;
#   - XDG_CONFIG_HOME and XDG_CACHE_HOME in the sandbox; PATH starts with stand-ins for sudo and nvidia-smi.
# It refuses to run without the sandbox marker, and the command does not run unless the checks inside hold.
set -u
SB=/home/thc1006/rebuild-rehearsal; R=$SB/root/home/thc1006
MEM=2G; CPUS=7; NAME=rr-$$; SYSBUS=0; HOMEBIND=0
while [ $# -gt 0 ]; do
  case $1 in
    -m) MEM=$2; shift 2 ;; -c) CPUS=$2; shift 2 ;; -n) NAME=$2; shift 2 ;; --sysbus) SYSBUS=1; shift ;; --home) HOMEBIND=1; shift ;;
    --) shift; break ;; *) break ;;
  esac
done
[ $# -gt 0 ] || { echo "usage: rr [-m MEM] [-c CPUS] [-n NAME] [--sysbus] -- COMMAND..." >&2; exit 2; }
[ -e "$R/dev/.rehearsal-sandbox" ] || { echo "rr: no sandbox marker in $R/dev" >&2; exit 99; }
case $NAME in rr-*) ;; *) echo "rr: the scope name must start with rr-" >&2; exit 2 ;; esac
bus=(--bind "$SB/root/empty-dbus" /run/dbus)
[ $SYSBUS = 1 ] && bus=()
binds=(--bind "$SB" "$SB" --ro-bind /home/thc1006/dev "$SB/prod/dev" --ro-bind /home/thc1006/cuda-12.9 "$SB/prod/cuda-12.9"
       --bind "$R/dev" /home/thc1006/dev --bind "$R/cuda-12.9" /home/thc1006/cuda-12.9 --bind "$R/downloads" /home/thc1006/downloads)
[ $HOMEBIND = 1 ] && binds=(--bind "$R" /home/thc1006 --bind "$SB" "$SB" --ro-bind /home/thc1006/dev "$SB/prod/dev"
       --ro-bind /home/thc1006/cuda-12.9 "$SB/prod/cuda-12.9")
exec systemd-run --user --scope -q --unit="$NAME" --description="rebuild rehearsal job $NAME" -p MemoryMax="$MEM" -p MemorySwapMax=0 -- \
  choom -n 500 -- nice -n 19 chrt -i 0 ionice -c3 taskset -c "$CPUS" \
  bwrap --die-with-parent --new-session --unshare-pid \
    --ro-bind / / --dev /dev --proc /proc \
    "${binds[@]}" --bind "$SB/root/tmp" /tmp \
    --bind "$SB/root/empty-run-user" /run/user/1003 "${bus[@]}" \
    --unsetenv DBUS_SESSION_BUS_ADDRESS --setenv XDG_RUNTIME_DIR /run/user/1003 \
    --setenv XDG_CONFIG_HOME "$R/.config" --setenv XDG_CACHE_HOME "$R/.cache" --setenv TMPDIR /tmp \
    --setenv PATH "$SB/shim:/usr/local/bin:/usr/bin:/bin" --setenv PYTHONNOUSERSITE 1 \
    /bin/sh -c 'if [ -e /home/thc1006/dev/.rehearsal-sandbox ] && [ ! -w /etc ] && [ ! -w "$0" ] && [ ! -w "$0/../cuda-12.9" ] \
                   && ! ls /dev/nvidia* >/dev/null 2>&1; then shift; exec "$@"; fi
                echo "rr: namespace check failed: nothing run" >&2; exit 98' "$SB/prod/dev" sh "$@"
[exit 0]
