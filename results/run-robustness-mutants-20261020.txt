"""Broken copies of bench/run_robustness.sh against tests/run_robustness_stubs.sh: each copy has one deliberate bug,
made by a text replacement below, and the harness must fail on it.  Usage: python3 -B mutate_robust.py <worktree>"""
import difflib, hashlib, re, subprocess, sys, tempfile
from pathlib import Path

root = Path(sys.argv[1])
src = (root / "bench/run_robustness.sh").read_text()
before = subprocess.run(["git", "-C", str(root), "show", "HEAD:bench/run_robustness.sh"],
                        capture_output=True, text=True, check=True).stdout


def rep(old, new, text=None):
    text = src if text is None else text
    assert text.count(old) == 1, old
    return text.replace(old, new)


mutants = {
    "the script before the fix (HEAD)": before,
    "sed without -u": rep('redact() { sed -u "${redact_args[@]}"; }', 'redact() { sed "${redact_args[@]}"; }'),
    "no first labels": rep("""names=$({ printf '%s\\n' "$names"; printf '%s\\n' "$names" | cut -d. -f1 | awk 'length >= 4'; } | sort -u |""",
                           """names=$({ printf '%s\\n' "$names"; } | sort -u |"""),
    "case-sensitive": rep('/<host>/gI"', '/<host>/g"'),
    "shortest name first": rep("awk '{print length, $0}' | sort -rn", "awk '{print length, $0}' | sort -n"),
    "no refusal without a name": rep("""[ -n "$names" ] || { echo "refusing: no host name to redact"; exit 1; }\n""", ""),
    "the journal around the redaction": rep('} 2>&1 | redact > "$LOG"', '} 3>&1 2>&1 | redact > "$LOG"',
                                            rep("  journalctl -u strata-q3.service", "  journalctl >&3 -u strata-q3.service")),
}
start, end = src.index("# the log holds lines of the journal"), src.index("START_LINE=$(wc -l")
mutants["the names found after the monitor started"] = rep(
    "trap 'kill $MON 2>/dev/null' EXIT\n", "trap 'kill $MON 2>/dev/null' EXIT\n" + src[start:end], src[:start] + src[end:])

def sha(p):
    return hashlib.sha256((root / p).read_bytes()).hexdigest()

ver = lambda *c: subprocess.run(c, capture_output=True, text=True).stdout.split("\n")[0]
print("# bench/run_robustness.sh against broken copies of itself, 2026-10-20, after the fix that writes the host name as")
print("# <host> in its log (the script and its harness as committed with the fix).")
print("#")
print("# Each copy has one deliberate bug, made by the text replacements in the runner at the end of this file; its diff")
print("# against the script is below.  tests/run_robustness_stubs.sh runs the copy in its sandbox; \"killed\" means that at")
print("# least one of its checks failed, and the checks that failed are listed.  The copy that passes looks for the host's")
print("# names after it starts the monitor: when it then refuses, its EXIT trap stops the monitor, as the script's does.")
print(f"# Machine: <host>, {ver('lsb_release', '-ds')}, {ver('python3', '-V')}, bash {ver('bash', '-c', 'echo $BASH_VERSION')}.  "
      "Temporary directories are written <tmp>.")

killed = 0
out = []
for name, text in mutants.items():
    with tempfile.TemporaryDirectory() as d:
        (Path(d) / "bench").mkdir()
        (Path(d) / "bench/run_robustness.sh").write_text(text)
        r = subprocess.run(["bash", str(root / "tests/run_robustness_stubs.sh"), d], capture_output=True, text=True,
                           timeout=300)
    fails = [re.sub(r"/tmp/robust\.\w+", "<tmp>", l.splitlines()[0])[:150] for l in r.stdout.splitlines() if l.startswith("FAIL ")]
    killed += r.returncode != 0
    out.append(f"\n== {name}: {'killed' if r.returncode else 'SURVIVED'}, {len(fails)} failed check{'' if len(fails) == 1 else 's'}")
    out += fails
    out.append("".join(difflib.unified_diff(src.splitlines(True), text.splitlines(True), "bench/run_robustness.sh",
                                            f"copy: {name}")).rstrip("\n"))
print(f"\n== {killed} of {len(mutants)} killed   (python3 -B mutate_robust.py <worktree>)")
print("\n".join(out))
print("\n== mutate_robust.py")
