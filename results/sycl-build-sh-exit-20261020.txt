## The script that ran it (re2990/run.sh)
    #!/bin/bash
    # #2990 again, with sycl/tools/build.sh of Strata 0.2.0-beta (3d1fb48), as on 2026-10-18: two small CMake projects with
    # one target named strata (Ninja, g++ 11.4), REPO and BUILD_DIR set to each; cmake and ninja of the Strata .venv on PATH.
    set -u
    here=$(cd "$(dirname "$0")" && pwd)
    W=$(mktemp -d)
    export PATH=$HOME/dev/strata/.venv/bin:$PATH
    for kind in good bad; do
      mkdir -p "$W/src-$kind"
      printf 'cmake_minimum_required(VERSION 3.16)\nproject(t CXX)\nadd_executable(strata main.cpp)\n' > "$W/src-$kind/CMakeLists.txt"
      if [ $kind = good ]; then echo 'int main() { return 0; }' > "$W/src-$kind/main.cpp"
      else echo 'int main() { return x; }' > "$W/src-$kind/main.cpp"; fi
      cmake -S "$W/src-$kind" -B "$W/b-$kind" -G Ninja > /dev/null 2>&1 || { echo "configure of $kind failed"; exit 1; }
    done
    echo "# sha256 of the script: $(sha256sum < "$here/build.sh" | cut -c1-64)"
    for kind in good bad; do
      echo "### $kind build:"
      REPO=$W BUILD_DIR=$W/b-$kind bash "$here/build.sh" strata
      echo "### build.sh exit status: $?"
    done 2>&1 | sed "s#$W#<work>#g"
    rm -rf "$W"
